function onEdit(e) { var sheet = e.source.getActiveSheet(); var range = e.range; // Aba Winners → envia perdedor para Losers if (sheet.getName() === "Winners" && range.getColumn() === 7) { // Coluna G = Vencedor var row = range.getRow(); if (row < 2) return; // ignora cabeçalho var roboA = sheet.getRange(row, 2).getValue(); // Coluna B var roboB = sheet.getRange(row, 4).getValue(); // Coluna D var vencedor = sheet.getRange(row, 7).getValue(); // Coluna G var perdedor = (vencedor === roboA) ? roboB : roboA; if (perdedor) { var losersSheet = e.source.getSheetByName("Losers"); losersSheet.appendRow([row, perdedor, "", "", "", "", ""]); } } // Aba Quartas → envia vencedores para Semifinal if (sheet.getName() === "Quartas de Final" && range.getColumn() === 7) { var row = range.getRow(); if (row < 2) return; var vencedor = sheet.getRange(row, 7).getValue(); if (vencedor) { var semiSheet = e.source.getSheetByName("Semifinal"); semiSheet.appendRow([row, vencedor, "", "", "", "", ""]); } } // Aba Semifinal → envia vencedores para Final if (sheet.getName() === "Semifinal" && range.getColumn() === 7) { var row = range.getRow(); if (row < 2) return; var vencedor = sheet.getRange(row, 7).getValue(); if (vencedor) { var finalSheet = e.source.getSheetByName("Final"); finalSheet.appendRow([row, vencedor, "", "", "", "", ""]); } } }
